<script>
document.addEventListener('DOMContentLoaded', function() {
	const table = document.querySelector('.post-content table');
	const rows = table
		? Array.from(table.querySelectorAll('tr')).filter(function(row) {
			return row.querySelectorAll('td').length > 0;
		})
		: [];
	const filters = Array.from(document.querySelectorAll('.pub-filter'));

	function updateFilterLabels() {
		const counts = {
			all: rows.length,
			journal: 0,
			proceedings: 0,
			book: 0,
			'under-review': 0,
		};

		rows.forEach(function(row) {
			const kind = row.getAttribute('data-kind') || 'journal';
			if (Object.prototype.hasOwnProperty.call(counts, kind)) {
				counts[kind] += 1;
			}
		});

		filters.forEach(function(button) {
			const key = button.getAttribute('data-filter') || 'all';
			const label = button.getAttribute('data-label') || button.textContent;
			const count = counts[key] || 0;
			button.textContent = label + ': ' + String(count);
		});
	}

	rows.forEach(function(row) {
		const cells = row.querySelectorAll('td');
		const journalText = (cells[2] ? cells[2].textContent : '').toLowerCase();

		let kind = 'journal';
		if (journalText.includes('under review')) {
			kind = 'under-review';
		} else if (journalText.includes('chapter')) {
			kind = 'book';
		} else if (journalText.includes('proceedings') || journalText.includes('iccs') || journalText.includes('archives')) 
        {
			kind = 'proceedings';
		}

		row.setAttribute('data-kind', kind);
	});

	filters.forEach(function(button) {
		button.addEventListener('click', function() {
			const selected = button.getAttribute('data-filter');

			filters.forEach(function(b) {
				b.classList.toggle('is-active', b === button);
			});

			rows.forEach(function(row) {
				const kind = row.getAttribute('data-kind');
				row.style.display = (selected === 'all' || selected === kind) ? '' : 'none';
			});
		});
	});

	updateFilterLabels();

	document.querySelectorAll('.post-content table td:nth-child(4) a:not(.bib-copy)').forEach(function(link) {
		link.setAttribute('target', '_blank');
		link.setAttribute('rel', 'noopener noreferrer');
	});

	document.querySelectorAll('.bib-copy').forEach(function(button) {
		const defaultTooltip = button.getAttribute('data-tooltip');

		button.addEventListener('click', function(event) {
			event.preventDefault();

			const bibtex = button.getAttribute('data-bibtex') || '';
			const showCopied = function() {
				button.setAttribute('data-tooltip', 'Copied!');
				button.classList.add('is-copied');
				setTimeout(function() {
					button.setAttribute('data-tooltip', defaultTooltip);
					button.classList.remove('is-copied');
				}, 1500);
			};

			if (navigator.clipboard && navigator.clipboard.writeText) {
				navigator.clipboard.writeText(bibtex).then(showCopied);
			} else {
				const textarea = document.createElement('textarea');
				textarea.value = bibtex;
				textarea.style.position = 'fixed';
				textarea.style.opacity = '0';
				document.body.appendChild(textarea);
				textarea.select();
				document.execCommand('copy');
				document.body.removeChild(textarea);
				showCopied();
			}
		});
	});

	if (typeof feather !== 'undefined') {
		feather.replace();
	}
});
</script>
